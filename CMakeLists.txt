cmake_minimum_required(VERSION 3.21)
set(ECLIPSE_INFER_VERSION "0.1.0")
if(DEFINED ENV{GITHUB_REF_NAME})
  string(REGEX REPLACE "^v" "" ECLIPSE_INFER_VERSION "$ENV{GITHUB_REF_NAME}")
endif()
project(EclipseInfer VERSION ${ECLIPSE_INFER_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(EclipseInfer src/main.cpp src/mainwindow.h src/mainwindow.cpp src/providerclient.h src/providerclient.cpp src/modelmanager.h src/modelmanager.cpp src/updater.h src/updater.cpp resources.qrc)
target_link_libraries(EclipseInfer PRIVATE Qt6::Widgets Qt6::Network)
target_compile_definitions(EclipseInfer PRIVATE ECLIPSE_INFER_VERSION=\"${PROJECT_VERSION}\")
if(WIN32)
  enable_language(RC)
  target_link_libraries(EclipseInfer PRIVATE Advapi32)
  # The resource compiler runs from the build directory, so give it the
  # project root as an include/search path for assets referenced by app.rc.
  target_include_directories(EclipseInfer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_sources(EclipseInfer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/app.rc")
  set_target_properties(EclipseInfer PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
