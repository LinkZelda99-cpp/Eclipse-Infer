cmake_minimum_required(VERSION 3.21)
set(ECLIPSE_ORBIT_VERSION "0.1.0")
if(DEFINED ENV{GITHUB_REF_NAME})
  string(REGEX REPLACE "^v" "" ECLIPSE_ORBIT_VERSION "$ENV{GITHUB_REF_NAME}")
endif()
project(EclipseOrbit VERSION ${ECLIPSE_ORBIT_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Multimedia)
qt_add_executable(EclipseOrbit src/main.cpp src/mainwindow.h src/mainwindow.cpp src/providerclient.h src/providerclient.cpp src/voiceclient.h src/voiceclient.cpp src/modelmanager.h src/modelmanager.cpp src/updater.h src/updater.cpp resources.qrc)
target_link_libraries(EclipseOrbit PRIVATE Qt6::Widgets Qt6::Network Qt6::Multimedia)
target_compile_definitions(EclipseOrbit PRIVATE ECLIPSE_ORBIT_VERSION=\"${PROJECT_VERSION}\")
if(WIN32)
  enable_language(RC)
  target_link_libraries(EclipseOrbit PRIVATE Advapi32)
  # The resource compiler runs from the build directory, so give it the
  # project root as an include/search path for assets referenced by app.rc.
  target_include_directories(EclipseOrbit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_sources(EclipseOrbit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/app.rc")
  set_target_properties(EclipseOrbit PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
